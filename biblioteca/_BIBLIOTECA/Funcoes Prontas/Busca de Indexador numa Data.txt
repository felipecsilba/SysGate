// Para índice acumulado mês a mês (ex. INPC em Débitos Gerais), usar:
// Utilitarios.buscaMovimentacoesDoIndexadorPorSiglaEmUmPeriodo('INPC', dataInicio, dataFim)
// que retorna uma lista de movimentações — ver "Calculo de Acrescimos" pra
// como acumular (Variante B).

// =========================================================
// LINGUAGEM DE SCRIPT (Scripts/Scripts, Fontes Dinamicas, Criticas) —
// não tem Utilitarios.*, usar a fonte de dados diretamente
// =========================================================
fonteIndexadores = Dados.tributos.v2.indexadores.movimentacoes   // ATENÇÃO: não é Dados.tributos.v2.indexadores (esse é só o cadastro do indexador, não as movimentações — dá erro "Invalid property <indexador.sigla>")

def buscarValorIndexador(sigla, dataReferencia) {
  def dado = fonteIndexadores.busca(
    criterio: "indexador.sigla = '${sigla}' and dtMovimentacao <= '${dataReferencia}'",
    ordem: "dtMovimentacao desc",
    primeiro: true
  )
  return dado?.vlMovimentacao ?: 0
}
